% Compact, implementation-specific algorithms for the main manuscript.
\begin{algorithm}[!t]
\caption{Matched preparation, selection, and refit}
\label{alg:matched-training}
\footnotesize
\begin{algorithmic}[1]
\Require Task $(r,C,H)$; site archives; frozen cuts and settings
\Ensure Origin-aligned forecasts and site-first metrics
\State Load the archive for $r$; audit the common time grid and GHI
\State For monthly tasks, average audited daily values by civil month
\State Build $\mathcal W_A,\mathcal W_V,\mathcal W_R,\mathcal W_T$
\Statex \hspace{\algorithmicindent}with context $y_{t-C:t-1}$ and targets $y_{t:t+H-1}$
\State Reject target windows crossing a cut; verify origin counts
\State Fit site scales $S_A$ before 2023 and $S_R$ before 2024
\If{hourly extension}
  \State Verify hashes; load the frozen official forecasts
  \State $\mathcal M\gets\{\mathrm{DilatedRNN}\}$; $\mathcal S\gets\{42\}$
\Else
  \State $\mathcal M\gets\{\mathrm{XGBoost,LSTM,TimesNet,DilatedRNN}\}$
  \State $\mathcal S\gets\{11,23,42,67,89\}$; evaluate causal baselines
\EndIf
\For{each $(m,s)\in\mathcal M\times\mathcal S$}
  \If{$m$ is neural}
    \State Initialize with $s$; fit $S_A(\mathcal W_A)$ using Adam
    \State Select $E^\star$ by early stopping on $S_A(\mathcal W_V)$ MSE
    \State Discard weights; reinitialize with the same seed $s$
    \State Fit $S_R(\mathcal W_R)$ for exactly $E^\star$ epochs
  \Else
    \State Fit separate fixed-budget XGBoost instances on $A$ and $R$
  \EndIf
  \State Forecast every $S_R(\mathcal W_T)$ context with fixed weights
  \Statex \hspace{\algorithmicindent}(Algorithm~\ref{alg:direct-architectures} for TimesNet/DilatedRNN)
  \State Invert $S_R$; retain the unbounded seed forecast in W/m$^2$
\EndFor
\State For daily/monthly tasks, average raw forecasts across seeds
\State Apply the zero floor and, if hourly, the nighttime mask
\State Verify common $(\ell,t,\mathrm{lead},\mathrm{target})$ keys
\State Compute site metrics, then equal-site macro summaries
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[!t]
\caption{Direct TimesNet and DilatedRNN forward passes}
\label{alg:direct-architectures}
\footnotesize
\begin{algorithmic}[1]
\Require Model $m$; fitted weights; scaled context $\mathbf x$; site $\ell$
\Ensure $H$ predictions in the external site-normalized scale
\If{$m=\mathrm{TimesNet}$}
  \State $\mu\gets\operatorname{mean}(\mathbf x)$;
  $\sigma\gets\sqrt{\operatorname{var}_{\mathrm{pop}}(\mathbf x)+10^{-5}}$
  \State $\mathbf u\gets(\mathbf x-\mu)/\sigma$
  \State $\mathbf Z\gets\mathrm{ValueEmbed}(\mathbf u)+\mathrm{PositionEmbed}$
  \State $\mathbf Z\gets\mathbf Z+\mathrm{SiteEmbed}(\ell)$
  \State Apply embedding dropout during training only
  \For{each TimesBlock}
    \State Compute FFT amplitudes separately for this sample
    \State Select the top-$k$ nonzero frequencies $f_i$
    \For{each selected $f_i$}
      \State $p_i\gets\lfloor C/f_i\rfloor$
      \State Pad and fold $\mathbf Z$ into cycles of length $p_i$
      \State Apply the two-layer Inception convolution with GELU
      \State Unfold and trim to $C$ steps, obtaining $\mathbf U_i$
    \EndFor
    \State $\alpha_i\gets\mathrm{softmax}_i(\mathrm{FFTAmplitude}(f_i))$
    \State $\mathbf Z\gets\mathrm{LayerNorm}(\mathbf Z+\sum_i\alpha_i\mathbf U_i)$
  \EndFor
  \State Project the temporal axis $C\to H$, then channels to one
  \State \Return $\sigma\widehat{\mathbf u}+\mu$
\Else \Comment{DilatedRNN}
  \State $\mathbf Z\gets\mathbf x$
  \For{each dilation $d\in(1,2,4)$}
    \State Process residue chains $\mathbf Z_{j,j+d,\ldots}$, $0\leq j<d$
    \Statex \hspace{2\algorithmicindent}with a shared tanh RNN and zero initial states
    \State Reassemble all outputs in their original time order
  \EndFor
  \State $\mathbf v\gets[\mathbf Z_{\mathrm{last}};\mathrm{SiteEmbed}(\ell)]$
  \State \Return $\mathrm{Linear}_{16\to H}(\mathrm{ReLU}(\mathrm{Linear}(\mathbf v)))$
\EndIf
\end{algorithmic}
\end{algorithm}
