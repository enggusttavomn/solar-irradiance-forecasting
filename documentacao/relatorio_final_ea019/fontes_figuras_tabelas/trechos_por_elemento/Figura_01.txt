 146 | def distancia_haversine_km(
 147 |     lat1: float,
 148 |     lon1: float,
 149 |     lat2: float,
 150 |     lon2: float,
 151 | ) -> float:
 152 |     """Calcula a distancia de grande circulo entre dois pontos.
 153 | 
 154 |     A formula de Haversine considera a curvatura da Terra. Ela e usada para
 155 |     confirmar que o ponto de grade retornado pelo NSRDB esta proximo da fabrica.
 156 |     """
 157 |     # Raio medio da Terra recomendado pela International Union of Geodesy.
 158 |     raio_terra_km = 6371.0088
 159 | 
 160 |     # Funcoes trigonometricas de ``math`` trabalham com radianos.
 161 |     phi1 = math.radians(lat1)
 162 |     phi2 = math.radians(lat2)
 163 |     delta_phi = math.radians(lat2 - lat1)
 164 |     delta_lambda = math.radians(lon2 - lon1)
 165 |     # Termo intermediario da formula de Haversine.
 166 |     a = (
 167 |         math.sin(delta_phi / 2) ** 2
 168 |         + math.cos(phi1) * math.cos(phi2) * math.sin(delta_lambda / 2) ** 2
 169 |     )
 170 |     # O arco angular e convertido em distancia ao multiplicar pelo raio.
 171 |     return 2 * raio_terra_km * math.asin(math.sqrt(a))
